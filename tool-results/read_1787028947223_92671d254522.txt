     1→"use client";
     2→

     4→import { useRouter } from "next/navigation";
     5→import Image from "next/image";
     6→import { ChevronRight, Check, Truck, CreditCard, Banknote, MapPin, Loader2, ShoppingCart, Plus, Home, Navigation, ShieldCheck, AlertCircle, Map as MapIcon } from "lucide-react";
     7→import { cn } from "@/lib/utils";

     9→import { Button } from "@/components/ui/button";
    10→import { Input } from "@/components/ui/input";

    12→import { Textarea } from "@/components/ui/textarea";
    13→import { Separator } from "@/components/ui/separator";
    14→import { useCart } from "@/contexts/CartContext";
    15→import { useOrders, type OrderAddress, type PaymentMethod } from "@/contexts/OrdersContext";
    16→import { useAddresses } from "@/contexts/AddressContext";
    17→import { formatINR, isValidPincode } from "@/lib/utils";
    18→import { appToast } from "@/lib/toast";
    19→import { FREE_SHIPPING_THRESHOLD, COD_MAX_AMOUNT } from "@/lib/constants";
    20→import Link from "next/link";
    21→import { useAuth } from "@/contexts/AuthContext";
    22→import { MapLocationPicker } from "@/components/common/MapLocationPicker";
    23→
    24→const STEPS = ["Address", "Review", "Payment"] as const;
    25→
    26→export default function CheckoutPage() {
    27→  const router = useRouter();
    28→  const { items, subtotal, itemCount, clearCart } = useCart();
    29→  const { createOrder } = useOrders();
    30→  const { isAuthenticated, isLoading: authLoading } = useAuth();
    31→  const { addresses, isLoading: addressesLoading } = useAddresses();
    32→
    33→  const [step, setStep] = useState(0);
    34→  const [isPlacing, setIsPlacing] = useState(false);
    35→  const [selectedAddressId, setSelectedAddressId] = useState<string | null>(null);
    36→  const [showNewAddressForm, setShowNewAddressForm] = useState(false);
    37→
    38→  // Address form
    39→  const [address, setAddress] = useState<OrderAddress>({
    40→    fullName: "", phone: "", addressLine1: "", addressLine2: "", landmark: "",
    41→    city: "Sonipat", state: "Haryana", pincode: "",
    42→  });
    43→  const [addressErrors, setAddressErrors] = useState<Record<string, string>>({});
    44→
    45→  // GPS verification for checkout new address
    46→  const [gpsState, setGpsState] = useState<"idle" | "detecting" | "fetching" | "verified" | "failed">("idle");
    47→  const [gpsError, setGpsError] = useState("");
    48→  const [gpsCoords, setGpsCoords] = useState<{ lat: number; lng: number } | null>(null);
    49→  const gpsVerified = gpsState === "verified";
    50→  const [mapPickerOpen, setMapPickerOpen] = useState(false);
    51→
    52→  // Payment
    53→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>("razorpay");
    54→  const [notes, setNotes] = useState("");
    55→  // D1: couponDiscount is 0 until coupon feature is implemented
    56→  const [couponDiscount] = useState(0);
    57→
    58→  const shipping = subtotal >= FREE_SHIPPING_THRESHOLD ? 0 : 49;
    59→  // GST 18% on (subtotal - discount). Tax-exclusive model.
    60→  const taxableAmount = Math.max(0, subtotal - couponDiscount);
    61→  const tax = Math.round(taxableAmount * 0.18);
    62→  // Total MUST include tax — previously tax was dropped (18% revenue loss)
    63→  const total = taxableAmount + shipping + tax;
    64→
    65→  // Auto-select default address on mount
    66→  useEffect(() => {
    67→    if (addresses.length > 0 && !selectedAddressId) {
    68→      const defaultAddr = addresses.find((a) => a.isDefault) ?? addresses[0];
    69→      setSelectedAddressId(defaultAddr.id);
    70→      // Populate the address form with the default address
    71→      setAddress({
    72→        fullName: defaultAddr.fullName,
    73→        phone: defaultAddr.phone,
    74→        addressLine1: defaultAddr.houseNo,
    75→        addressLine2: defaultAddr.locality,
    76→        landmark: "",
    77→        city: defaultAddr.city,
    78→        state: defaultAddr.state,
    79→        pincode: defaultAddr.pincode,
    80→      });
    81→    }
    82→  }, [addresses, selectedAddressId]);
    83→
    84→  // Handle selecting a saved address — BYPASS FIX: only allow GPS-verified addresses
    85→  const handleSelectAddress = (addrId: string) => {
    86→    const addr = addresses.find((a) => a.id === addrId);
    87→    if (!addr) return;
    88→    // BYPASS FIX: Block selection of non-GPS-verified addresses
    89→    if (!addr.gpsVerified) {
    90→      appToast.warning("Address not verified", "This address doesn't have GPS verification. Please verify it in Address Book or add a new address.");
    91→      return;
    92→    }
    93→    setSelectedAddressId(addrId);
    94→    setShowNewAddressForm(false);
    95→    setAddress({
    96→      fullName: addr.fullName,
    97→      phone: addr.phone,
    98→      addressLine1: addr.houseNo,
    99→      addressLine2: addr.locality,
   100→      landmark: "",
   101→      city: addr.city,
   102→      state: addr.state,
   103→      pincode: addr.pincode,
   104→    });
   105→    setAddressErrors({});
   106→  };
   107→
   108→  // Handle "Add new address" click
   109→  const handleAddNewAddress = () => {
   110→    setSelectedAddressId(null);
   111→    setShowNewAddressForm(true);
   112→    setAddress({
   113→      fullName: "", phone: "", addressLine1: "", addressLine2: "", landmark: "",
   114→      city: "Sonipat", state: "Haryana", pincode: "",
   115→    });
   116→    setAddressErrors({});
   117→    setGpsState("idle");
   118→    setGpsError("");
   119→  };
   120→
   121→  // GPS verification handler for checkout new address
   122→  const handleGPS = async () => {
   123→    setGpsState("detecting");
   124→    setGpsError("");
   125→    try {
   126→      if (!navigator.geolocation) {
   127→        setGpsState("failed");
   128→        setGpsError("GPS not supported on this device");
   129→        return;
   130→      }
   131→      navigator.geolocation.getCurrentPosition(
   132→        async (pos) => {
   133→          const { latitude, longitude, accuracy } = pos.coords;
   134→          if (accuracy > 100) {
   135→            setGpsState("failed");
   136→            setGpsError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   137→            return;
   138→          }
   139→          setGpsState("fetching");
   140→          try {
   141→            const url = `https://nominatim.openstreetmap.org/reverse?format=json&lat=${latitude}&lon=${longitude}&addressdetails=1`;
   142→            const res = await fetch(url, {
   143→              headers: { "Accept-Language": "en", "User-Agent": "GrowPlants/1.0 (hello@growplants.in)" },
   144→            });
   145→            if (!res.ok) throw new Error("Reverse geocoding failed");
   146→            const data = await res.json();
   147→            const addr = data.address || {};
   148→            setAddress((prev) => ({
   149→              ...prev,
   150→              city: addr.city || addr.town || addr.village || addr.county || prev.city,
   151→              state: addr.state || prev.state,
   152→              pincode: addr.postcode || prev.pincode,
   153→            }));
   154→            setGpsState("verified");
   155→            appToast.success("Location verified!", `Accuracy: ${Math.round(accuracy)}m`);
   156→          } catch (geoErr) {
   157→            setGpsState("failed");
   158→            setGpsError("Could not fetch address from GPS. Please enter manually.");
   159→          }
   160→        },
   161→        (err) => {
   162→          setGpsState("failed");
   163→          if (err.code === 1) setGpsError("Location permission denied. Please allow location access.");
   164→          else if (err.code === 2) setGpsError("Location unavailable. Check your GPS settings.");
   165→          else if (err.code === 3) setGpsError("Location request timed out. Try again.");
   166→          else setGpsError("Failed to get location.");
   167→        },
   168→        { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
   169→      );
   170→    } catch (err) {
   171→      setGpsState("failed");
   172→      setGpsError(err instanceof Error ? err.message : "GPS verification failed");
   173→    }
   174→  };
   175→
   176→  // Reset GPS when switching to a saved address
   177→  useEffect(() => {
   178→    if (selectedAddressId) {
   179→      setGpsState("idle");
   180→      setGpsError("");
   181→      setGpsCoords(null);
   182→    }
   183→  }, [selectedAddressId]);
   184→
   185→  // Handle manual location selection from map
   186→  const handleMapLocationSelect = (location: { lat: number; lng: number; accuracy: number; city?: string; state?: string; pincode?: string }) => {
   187→    setGpsCoords({ lat: location.lat, lng: location.lng });
   188→    setAddress((prev) => ({
   189→      ...prev,
   190→      city: location.city || prev.city,
   191→      state: location.state || prev.state,
   192→      pincode: location.pincode || prev.pincode,
   193→      latitude: location.lat,
   194→      longitude: location.lng,
   195→    }));
   196→    setGpsState("verified");
   197→    setGpsError("");
   198→    appToast.success("Location set!", "Pin location confirmed on map");
   199→  };
   200→
   201→  // BYPASS FIX: Track GPS-verified values — if user edits city/state/pincode after GPS verify, reset
   202→  const gpsVerifiedRef = useRef<{ city: string; state: string; pincode: string } | null>(null);
   203→  useEffect(() => {
   204→    if (gpsState === "verified") {
   205→      gpsVerifiedRef.current = { city: address.city, state: address.state, pincode: address.pincode };
   206→    }
   207→  }, [gpsState]); // intentionally not depending on address to avoid re-trigger
   208→
   209→  useEffect(() => {
   210→    // If GPS was verified and user changes city/state/pincode, reset GPS
   211→    if (gpsState === "verified" && gpsVerifiedRef.current) {
   212→      if (
   213→        gpsVerifiedRef.current.city !== address.city ||
   214→        gpsVerifiedRef.current.state !== address.state ||
   215→        gpsVerifiedRef.current.pincode !== address.pincode
   216→      ) {
   217→        setGpsState("idle");
   218→        gpsVerifiedRef.current = null;
   219→      }
   220→    }
   221→  }, [address.city, address.state, address.pincode, gpsState]);
   222→
   223→  // C3 FIX: Auth guard — redirect to login if not authenticated
   224→  useEffect(() => {
   225→    if (!authLoading && !isAuthenticated) {
   226→      router.replace("/login?redirect=/checkout");
   227→    }
   228→  }, [authLoading, isAuthenticated, router]);
   229→
   230→  // C2 FIX: COD max amount enforcement
   231→  const codExceedsLimit = paymentMethod === "cod" && total > COD_MAX_AMOUNT;
   232→
   233→  // Empty cart guard
   234→  if (itemCount === 0 && !isPlacing) {
   235→    return (
   236→      <Container className="py-16">
   237→        <div className="text-center space-y-4">
   238→          <div className="size-16 rounded-full bg-[#F3F8F1] flex items-center justify-center mx-auto">
   239→            <ShoppingCart className="size-8 text-[#1A6B3C]" aria-hidden="true" />
   240→          </div>
   241→          <h1 className="text-xl font-bold text-gray-900">Your cart is empty</h1>
   242→          <p className="text-sm text-slate-500">Add some products before checking out.</p>
   243→          <Button asChild className="bg-[#1A6B3C] hover:bg-[#16A34A] gap-2"><Link href="/shop">Browse Shop</Link></Button>
   244→        </div>
   245→      </Container>
   246→    );
   247→  }
   248→
   249→  const validateAddress = () => {
   250→    const errs: Record<string, string> = {};
   251→    if (!address.fullName.trim()) errs.fullName = "Name is required";
   252→    if (!address.phone.trim() || !/^[6-9]\d{9}$/.test(address.phone.replace(/\D/g, ""))) errs.phone = "Enter a valid 10-digit phone number";
   253→    if (!address.addressLine1.trim()) errs.addressLine1 = "Address is required";
   254→    if (!address.city.trim()) errs.city = "City is required";
   255→    if (!address.state.trim()) errs.state = "State is required";
   256→    if (!isValidPincode(address.pincode)) errs.pincode = "Enter a valid 6-digit pincode";
   257→    // BYPASS FIX: Require GPS verification for new addresses (not for saved addresses)
   258→    if (showNewAddressForm && !selectedAddressId && !gpsVerified) {
   259→      errs.gps = "GPS verification is required for new addresses";
   260→    }
   261→    setAddressErrors(errs);
   262→    return Object.keys(errs).length === 0;
   263→  };
   264→
   265→  const handleNext = () => {
   266→    if (step === 0 && !validateAddress()) return;
   267→    setStep((s) => Math.min(s + 1, STEPS.length - 1));
   268→  };
   269→
   270→  const handlePlaceOrder = async () => {
   271→    // C2 FIX: Enforce COD max amount
   272→    if (paymentMethod === "cod" && total > COD_MAX_AMOUNT) {
   273→      appToast.error("COD limit exceeded", `Cash on Delivery is only available for orders up to ₹${COD_MAX_AMOUNT}. Please choose online payment.`);
   274→      return;
   275→    }
   276→    setIsPlacing(true);
   277→    try {
   278→      const order = await createOrder({
   279→        items: items.map((i) => ({ productId: i.productId, name: i.name, slug: i.slug, price: i.price, image: i.image, quantity: i.quantity, variantId: i.variantId })),
   280→        subtotal, shipping, discount: couponDiscount, tax, total,
   281→        address, paymentMethod, notes: notes.trim() || undefined,
   282→      });
   283→
   284→      // Only clear cart + redirect on genuine success (createOrder throws on failure)
   285→      clearCart();
   286→      appToast.success("Order placed!", `Order ${order.orderNumber} confirmed`);
   287→      router.push(`/order-confirmation/${order.id}`);
   288→      // D3: isPlacing is NOT reset — page navigates away, so it's fine
   289→      // (the next page mount will have a fresh state)
   290→    } catch (err) {
   291→      console.error("[checkout] createOrder failed:", err);
   292→      // A7 FIX: Show the actual error message from the API; cart is NOT cleared
   293→      const message = err instanceof Error ? err.message : "Could not place your order. Please try again.";
   294→      appToast.error("Order failed", message);
   295→      setIsPlacing(false);
   296→    }
   297→  };
   298→
   299→  return (
   300→    <Container className="py-6 md:py-10">
   301→      {/* Stepper */}
   302→      <div className="flex items-center justify-center gap-2 mb-8">
   303→        {STEPS.map((label, i) => (
   304→          <div key={label} className="flex items-center gap-2 animate-scale-in" style={{ animationDelay: `${i * 80}ms` }}>
   305→            <div className={cn("flex items-center gap-2 px-3 py-1.5 rounded-full text-sm font-medium transition-colors", i === step ? "bg-[#1A6B3C] text-white" : i < step ? "bg-green-100 text-green-700" : "bg-slate-100 text-slate-500")}>
   306→              {i < step ? <Check className="size-3.5" /> : <span className="size-5 rounded-full flex items-center justify-center text-xs">{i + 1}</span>}
   307→              {label}
   308→            </div>
   309→            {i < STEPS.length - 1 && <ChevronRight className="size-4 text-slate-300" />}
   310→          </div>
   311→        ))}
   312→      </div>
   313→
   314→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 lg:gap-8">
   315→        {/* Left: Step content */}
   316→        <div className="lg:col-span-2 space-y-4">
   317→          {/* Step 1: Address */}
   318→          {step === 0 && (
   319→            <div className="bg-white border border-slate-200 rounded-xl p-5 space-y-4">
   320→              <h2 className="text-base font-bold text-[#1A6B3C] flex items-center gap-2"><MapPin className="size-5" />Delivery Address</h2>
   321→              <Separator />
   322→
   323→              {/* Saved Addresses Section */}
   324→              {!addressesLoading && addresses.length > 0 && (
   325→                <div className="space-y-3">
   326→                  <p className="text-sm font-semibold text-slate-700">Saved Addresses</p>
   327→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   328→                    {addresses.map((addr) => (
   329→                      <button
   330→                        key={addr.id}
   331→                        type="button"
   332→                        onClick={() => handleSelectAddress(addr.id)}
   333→                        className={cn(
   334→                          "text-left p-4 rounded-xl border-2 transition-all relative",
   335→                          selectedAddressId === addr.id
   336→                            ? "border-[#1A6B3C] bg-[#F3F8F1] shadow-sm"
   337→                            : "border-slate-200 hover:border-[#1A6B3C]/30 bg-white",
   338→                          !addr.gpsVerified && "opacity-60 cursor-not-allowed hover:border-slate-200"
   339→                        )}
   340→                      >
   341→                        {selectedAddressId === addr.id && (
   342→                          <div className="absolute top-3 right-3 size-5 rounded-full bg-[#1A6B3C] flex items-center justify-center">
   343→                            <Check className="size-3 text-white" strokeWidth={3} />
   344→                          </div>
   345→                        )}
   346→                        <div className="flex items-start gap-2 mb-2">
   347→                          <div className="size-8 rounded-lg bg-[#1A6B3C]/10 flex items-center justify-center shrink-0">
   348→                            <Home className="size-4 text-[#1A6B3C]" />
   349→                          </div>
   350→                          <div className="min-w-0 flex-1 pr-6">
   351→                            <p className="text-sm font-bold text-slate-800 truncate">
   352→                              {addr.fullName}
   353→                              {addr.isDefault && (
   354→                                <span className="ml-2 text-[10px] font-semibold text-[#1A6B3C] bg-[#1A6B3C]/10 px-1.5 py-0.5 rounded-full">
   355→                                  Default
   356→                                </span>
   357→                              )}
   358→                            </p>
   359→                            <p className="text-xs text-slate-500">{addr.label} · {addr.phone}</p>
   360→                          </div>
   361→                        </div>
   362→                        <p className="text-xs text-slate-600 leading-relaxed">
   363→                          {addr.houseNo}, {addr.locality}, {addr.city}, {addr.state} - {addr.pincode}
   364→                        </p>
   365→                        {addr.gpsVerified ? (
   366→                          <p className="text-[10px] text-green-600 mt-1.5 flex items-center gap-1">
   367→                            <ShieldCheck className="size-2.5" /> GPS Verified
   368→                          </p>
   369→                        ) : (
   370→                          <p className="text-[10px] text-red-500 mt-1.5 flex items-center gap-1">
   371→                            <AlertCircle className="size-2.5" /> Not GPS Verified — tap to fix
   372→                          </p>
   373→                        )}
   374→                      </button>
   375→                    ))}
   376→
   377→                    {/* Add New Address Card */}
   378→                    <button
   379→                      type="button"
   380→                      onClick={handleAddNewAddress}
   381→                      className={cn(
   382→                        "p-4 rounded-xl border-2 border-dashed transition-all flex flex-col items-center justify-center gap-2 min-h-[120px]",
   383→                        showNewAddressForm
   384→                          ? "border-[#1A6B3C] bg-[#F3F8F1]"
   385→                          : "border-slate-300 hover:border-[#1A6B3C]/50 text-slate-500 hover:text-[#1A6B3C]"
   386→                      )}
   387→                    >
   388→                      <Plus className="size-6" />
   389→                      <span className="text-sm font-medium">Add New Address</span>
   390→                    </button>
   391→                  </div>
   392→                </div>
   393→              )}
   394→
   395→              {/* Loading state */}
   396→              {addressesLoading && (
   397→                <div className="space-y-2">
   398→                  <div className="h-4 bg-slate-100 rounded animate-pulse w-32" />
   399→                  <div className="h-24 bg-slate-100 rounded animate-pulse" />
   400→                </div>
   401→              )}
   402→
   403→              {/* New Address Form (shown when "Add New" clicked OR no saved addresses) */}
   404→              {(showNewAddressForm || addresses.length === 0) && (
   405→                <div className="space-y-3 pt-2">
   406→                  {addresses.length > 0 && (
   407→                    <p className="text-sm font-semibold text-slate-700">Enter New Address</p>
   408→                  )}
   409→
   410→                  {/* GPS Verification Button */}
   411→                  <div className="p-3 bg-[#F3F8F1] rounded-lg border border-[#1A6B3C]/10">
   412→                    <div className="flex items-center justify-between gap-2">
   413→                      <div className="flex items-center gap-2">
   414→                        {gpsVerified ? (
   415→                          <ShieldCheck className="size-5 text-green-600" />
   416→                        ) : gpsState === "detecting" || gpsState === "fetching" ? (
   417→                          <Loader2 className="size-5 text-[#1A6B3C] animate-spin" />
   418→                        ) : gpsState === "failed" ? (
   419→                          <AlertCircle className="size-5 text-red-500" />
   420→                        ) : (
   421→                          <Navigation className="size-5 text-[#1A6B3C]" />
   422→                        )}
   423→                        <div>
   424→                          <p className="text-sm font-medium text-slate-700">
   425→                            {gpsVerified ? "Location Verified" : gpsState === "detecting" ? "Detecting location..." : gpsState === "fetching" ? "Fetching address..." : "GPS Verification"}
   426→                          </p>
   427→                          <p className="text-xs text-slate-500">
   428→                            {gpsVerified ? "City, state, pincode auto-filled from GPS" : "Verify your delivery location (within 100m accuracy)"}
   429→                          </p>
   430→                        </div>
   431→                      </div>
   432→                      {!gpsVerified && gpsState !== "detecting" && gpsState !== "fetching" && (
   433→                        <Button type="button" size="sm" variant="outline" className="border-[#1A6B3C] text-[#1A6B3C] gap-1.5 animate-pulse-ring" onClick={handleGPS}>
   434→                          <Navigation className="size-3.5" /> Verify
   435→                        </Button>
   436→                      )}
   437→                    </div>
   438→                    {gpsError && <p className="text-xs text-red-500 mt-2">{gpsError}</p>}
   439→                    {addressErrors.gps && !gpsVerified && (
   440→                      <p className="text-xs text-red-500 mt-2 flex items-center gap-1">
   441→                        <AlertCircle className="size-3.5 shrink-0" />{addressErrors.gps}
   442→                      </p>
   443→                    )}
   444→                    {/* Set Location Manually — drag pin on map */}
   445→                    {!gpsVerified && (
   446→                      <div className="mt-2 pt-2 border-t border-slate-100">
   447→                        <button
   448→                          type="button"
   449→                          onClick={() => setMapPickerOpen(true)}
   450→                          className="text-xs text-[#1A6B3C] hover:underline font-medium flex items-center gap-1"
   451→                        >
   452→                          <MapIcon className="size-3.5" />
   453→                          Set Location Manually on Map
   454→                        </button>
   455→                      </div>
   456→                    )}
   457→                  </div>
   458→
   459→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   460→                    <div className="space-y-1.5"><Label htmlFor="fullName" className="text-sm">Full Name *</Label><Input id="fullName" value={address.fullName} onChange={(e) => setAddress({ ...address, fullName: e.target.value })} className="h-11" />{addressErrors.fullName && <p className="text-xs text-red-500">{addressErrors.fullName}</p>}</div>
